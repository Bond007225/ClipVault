# Builds the engine-independent gameplay core and its tests without Unreal.
cmake_minimum_required(VERSION 3.20)
project(CourtLifeCoreTests CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CORE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../Source/CourtLife/Core)
file(GLOB CORE_SOURCES ${CORE_DIR}/*.cpp)
add_executable(courtlife_core_tests CoreTests.cpp ${CORE_SOURCES})
target_include_directories(courtlife_core_tests PRIVATE ${CORE_DIR} ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(courtlife_core_tests PRIVATE /W4 /WX)
else()
  target_compile_options(courtlife_core_tests PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
endif()
